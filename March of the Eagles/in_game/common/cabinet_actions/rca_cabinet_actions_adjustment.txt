# Assimilate Province (promote_culture)
# Vanilla-engine action available to AI and player
# Promotes primary culture in a chosen province - all non-primary pops assimilate over time
# REPLACE owns: rate, allow threshold, religion group culture filter, unrest
REPLACE:promote_culture = {
	ability = dip

	allow_multiple = yes

	potential = {
		num_locations > 0
	}

	allow = {
		cultural_unity < 1.0
		# RCA: disallow if primary culture is below 5% of country population
		"culture_percentage_in_country(culture)" >= 0.05
	}

	select_trigger = {
		looking_for_a = culture
		target_flag = target
		source_ai_override = actor
		source_flags_ai_override = include_any_present
		name = "promote_culture_select_culture"
		none_available_msg_key = "promote_culture_no_culture"
		column = { data = name }
		column = { data = culture_percentage_in_country }
		default_sort = culture_percentage_in_country
		visible = {
			OR = {
				AND = { # for humans
					scope:actor = { is_ai = no }
					# RCA: only primary or accepted cultures
					is_primary_or_accepted_in = scope:actor
				}
				AND = { # for AI
					scope:actor = { is_ai = yes }
					is_primary_in = scope:actor
				}
			}
		}
	}

	select_trigger = {
		looking_for_a = province
		source = actor
		target_flag = target_1
		name = "promote_culture_select_province"
		none_available_msg_key = "promote_culture_no_provinces"
		column = { data = name }
		column = { data = province_culture_percentage }
		column = { data = population }
		default_sort = province_culture_percentage
		visible = {
			any_location_in_province = {
				OR = {
					NOT = { dominant_culture = owner.culture }
					local_cultural_unity < 1.0
				}
			}
			# Block if any assimilation action is already active on this province
			NOT = { cabinet_already_performing_same_task_on_target = { cabinet_action = promote_culture interaction_target = target_1 } }
		}
		map_color = {
			lerp = {
				min_color = define:NMapColors|MAP_COLOR_LOW
				max_color = define:NMapColors|MAP_COLOR_HIGH
				factor = "province.culture_percentage(scope:target)"
			}
		}
	}

	is_finished = {
		scope:target_1 = {
			"culture_percentage(scope:target)" >= 1.0
			not = {
				any_location_in_province = {
					NOT = { dominant_culture = scope:target }
				}
			}
		}
	}

	on_activate = {
		if = {
			limit = { exists = scope:target_1 }
			scope:target_1 = {
				set_variable = {
					name = promoting_culture
					value = scope:target
				}
			}
		}
	}

	on_deactivate = {
		if = {
			limit = { exists = scope:target_1 }
			scope:target_1 = {
				remove_variable = promoting_culture
			}
		}
	}

	location_modifier = {
		local_pop_assimilation_speed = 0.0035 # RCA: reduced from 0.04
		local_unrest = 0.1 # RCA: local unrest while active
		local_migration_attraction = -0.15 # 1.2: suppresses in-migration to target province while active
	}

	progress = {
		scope:target_1 = {
			value = "culture_percentage(scope:target)"
		}
	}

	map_marker = {
		show_for_owner = yes
	}

	ai_will_do = {
		add = {
			desc = "Wrong culture"
			if = {
				limit = {
					exists = scope:target
					scope:target != scope:actor.culture
				}
				value = -1000
			}
		}
	}
}

# Assimilate Area (assimilate_area)
# Vanilla-engine action available to AI and player
# Promotes primary culture across an entire area
# REPLACE owns: rate, allow threshold, religion group culture filter, unrest
REPLACE:assimilate_area = {
	ability = dip
	allow_multiple = no

	potential = {
		modifier:allow_cabinet_assimilate_area = yes
		num_locations > 0
	}

	allow = {
		cultural_unity < 1.0
		# RCA: disallow if primary culture is below 5% of country population
		"culture_percentage_in_country(culture)" >= 0.05
	}

	select_trigger = {
		looking_for_a = culture
		target_flag = target_culture
		source_ai_override = actor
		source_flags_ai_override = include_any_present
		name = "promote_culture_select_culture"
		none_available_msg_key = "promote_culture_no_culture"
		column = { data = name }
		column = { data = culture_percentage_in_country }
		default_sort = culture_percentage_in_country
		visible = {
			OR = {
				AND = { # for humans
					scope:actor = { is_ai = no }
					# RCA: only primary or accepted cultures
					is_primary_or_accepted_in = scope:actor
				}
				AND = { # for AI
					scope:actor = { is_ai = yes }
					is_primary_in = scope:actor
				}
			}
		}
	}

	select_trigger = {
		looking_for_a = area
		source = actor
		target_flag = target
		name = "assimilate_area_select_area"
		none_available_msg_key = "assimilate_area_no_areas"
		column = { data = name }
		column = { data = area_primary_culture_percentage }
		column = { data = population }
		default_sort = area_primary_culture_percentage
		visible = {
			any_location_in_area = {
				local_cultural_unity < 1.0
				owner ?= { this = scope:actor }
			}
			# 1.2: AI restricted to areas where they have meaningful control
			OR = {
				scope:actor = { is_ai = no }
				"area_average_control(scope:actor)" >= 0.5
			}
			# Block if any area assimilation action is already active on this area
			NOT = { cabinet_already_performing_same_task_on_target = { cabinet_action = assimilate_area interaction_target = root } }
		}
		map_color = {
			lerp = {
				min_color = define:NMapColors|MAP_COLOR_LOW
				max_color = define:NMapColors|MAP_COLOR_HIGH
				factor = "area.culture_percentage(scope:target_culture)"
			}
		}
	}

	on_activate = {
		if = {
			limit = { exists = scope:target }
			scope:target = {
				every_province_in_area = {
					limit = { owner = scope:actor }
					set_variable = {
						name = promoting_culture
						value = scope:target_culture
					}
				}
			}
		}
	}

	on_deactivate = {
		if = {
			limit = { exists = scope:target }
			scope:target = {
				every_province_in_area = {
					limit = { owner = scope:actor }
					remove_variable = promoting_culture
				}
			}
		}
	}

	location_modifier = {
		local_pop_assimilation_speed = 0.0035 # RCA: reduced from 0.02
		local_unrest = 0.1 # RCA: local unrest while active
		local_migration_attraction = -0.2 # 1.2: suppresses in-migration to target area while active
	}

	map_marker = {
		show_for_owner = yes
	}

	progress = {
		scope:target = {
			value = "culture_percentage_in_area(scope:actor|scope:target_culture)"
		}
	}
}

# Convert Province (promote_religion)
# Vanilla-engine action available to AI and player
# Converts pops in a chosen province toward a chosen religion within the actor's group
# REPLACE owns: rate, local unrest, migration attraction
REPLACE:promote_religion = {
	ability = mil

	allow_multiple = yes

	potential = {
		num_locations > 0
	}

	allow = {
		religious_unity < 1.0
		trigger_if = {
			limit = { has_country_modifier = sco_the_first_congregation }
			NOT = { religion = religion:catholic }
		}
		OR = {
			modifier:global_nobles_conversion_blocked = no
			modifier:global_burghers_conversion_blocked = no
			modifier:global_clergy_conversion_blocked = no
			modifier:global_peasants_conversion_blocked = no
		}
	}

	# Vanilla religion selector - retained; filtered to actor's religion group
	select_trigger = {
		looking_for_a = religion
		target_flag = target_religion
		source_ai_override = actor
		source_flags_ai_override = include_any_present
		name = "promote_religion_select_religion"
		none_available_msg_key = "convert_religion_no_religions"
		column = { data = name }
		column = { data = religion_percentage_in_country }
		default_sort = religion_percentage_in_country
		visible = {
			group = scope:actor.religion.group
			is_religion_enabled = yes
			OR = {
				scope:actor.religion = root
				scope:actor = {
					religion_percentage_in_country = {
						religion = root
						value >= 0.01
					}
				}
			}
		}
	}

	select_trigger = {
		looking_for_a = province
		source = actor
		target_flag = target_province
		name = "promote_religion_select_province"
		none_available_msg_key = "promote_religion_no_provinces"
		column = { data = name }
		column = { data = province_religion_percentage }
		column = { data = population }
		default_sort = province_religion_percentage
		visible = {
			any_location_in_province = {
				OR = {
					NOT = { dominant_religion = owner.religion }
					local_religious_unity < 1.0
				}
			}
			# Block if any conversion action is already active on this province
			NOT = { cabinet_already_performing_same_task_on_target = { cabinet_action = promote_religion interaction_target = target_province } }
		}
		map_color = {
			lerp = {
				min_color = define:NMapColors|MAP_COLOR_LOW
				max_color = define:NMapColors|MAP_COLOR_HIGH
				factor = "province.religion_percentage(scope:target_religion)"
			}
		}
	}

	is_finished = {
		scope:target_province = {
			"religion_percentage(scope:target_religion)" = 1.0
		}
	}

	on_activate = {
		if = {
			limit = { exists = scope:target_province }
			scope:target_province = {
				set_variable = {
					name = promoting_religion
					value = scope:target_religion
				}
			}
		}
	}

	on_deactivate = {
		if = {
			limit = { exists = scope:target_province }
			scope:target_province = {
				remove_variable = promoting_religion
			}
		}
	}

	location_modifier = {
		local_pop_conversion_speed = 0.007 # RCA: reduced from 0.04
		local_unrest = 0.1 # RCA: local unrest while active
		local_migration_attraction = -0.15 # 1.2: suppresses in-migration to target province while active
	}

	progress = {
		scope:target_province = {
			value = "religion_percentage(scope:target_religion)"
		}
	}

	map_marker = {
		show_for_owner = yes
	}

	ai_will_do = {
		add = {
			desc = "Wrong religion"
			if = {
				limit = {
					exists = scope:target_religion
					scope:target_religion != scope:actor.religion
				}
				value = -1000
			}
		}
	}
}
