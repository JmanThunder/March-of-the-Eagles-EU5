# lutheran_preachers and calvinist_preachers (situation_buildings)
# These buildings drive passive Reformation spread via conversion_religion = lutheran/calvinist
# and building_missionary_effort_scaled = 0.01/level/month — entirely bypassing the
# global_pop_conversion_speed system RCA zeroes out. Without adjustment, Lutheran/Calvinist
# spread at full vanilla rate while Catholic counter-conversion is at 17.5% of vanilla,
# creating a ~5x asymmetry that makes the Reformation unstoppable.
#
# Fix: local_pop_conversion_speed_modifier = -0.5 cuts building spread by 50%.
# This preserves the historical regional gradient based on distance from Rome
# (which determines building max level: 1 in Iberia, up to 4 in Scandinavia):
#
#   Iberia/Italy    (level 1): 0.005/mo spread — one cabinet (0.007) wins     [OTL: Catholic holds]
#   France/C.Europe (level 2): 0.010/mo spread — requires two cabinets         [OTL: slow recovery]
#   N.Germany       (level 3): 0.015/mo spread — two cabinets barely match     [OTL: stays Protestant]
#   Scandinavia     (level 4): 0.020/mo spread — effectively unreclaimable     [OTL: fully Lutheran]
#
# Both buildings get the same modifier so Lutheran and Calvinist behave symmetrically.
INJECT:lutheran_preachers = {
	modifier = {
		local_pop_conversion_speed_modifier = -0.5 # Halve spread rate; see balance note above
	}
}

INJECT:calvinist_preachers = {
	modifier = {
		local_pop_conversion_speed_modifier = -0.5 # Halve spread rate; matches lutheran_preachers
	}
}
